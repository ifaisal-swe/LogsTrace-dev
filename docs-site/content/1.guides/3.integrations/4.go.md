```go [logstrace.go]
package logstrace

import (
	"bytes"
	"context"
	"encoding/json"
	"net/http"
	"os"
	"time"
)

type Event struct {
	Text    string         `json:"text"`
	Channel string         `json:"channel,omitempty"`
	Level   string         `json:"level,omitempty"`
	TS      int64          `json:"ts,omitempty"`
	TraceID string         `json:"traceId,omitempty"`
	Data    map[string]any `json:"data,omitempty"`
}

var client = &http.Client{Timeout: 5 * time.Second}

// Send posts up to 500 events in one request.
func Send(ctx context.Context, events ...Event) error {
	body, err := json.Marshal(map[string]any{"events": events})
	if err != nil {
		return err
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, "https://logstrace.dev/api/ingest", bytes.NewReader(body))
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer "+os.Getenv("LOGSTRACE_KEY"))
	req.Header.Set("Content-Type", "application/json")
	res, err := client.Do(req)
	if err != nil {
		return err
	}
	return res.Body.Close()
}
```

```go
logstrace.Send(ctx, logstrace.Event{

})
```

::tip
Call `Send` from a goroutine fed by a channel so request handlers never wait on logging.
::
