Works with the standard `logging` module. Needs `requests`.

```python [logstrace.py]
import atexit, logging, os, queue, threading, time
import requests

URL = "https://logstrace.dev/api/ingest"


class LogsTraceHandler(logging.Handler):
    def __init__(self, key=None, channel=None, flush_every=1.0):
        super().__init__()
        self.session = requests.Session()
        self.session.headers["Authorization"] = f"Bearer {key or os.environ['LOGSTRACE_KEY']}"
        self.channel = channel
        self.q = queue.Queue(maxsize=10_000)
        self.flush_every = flush_every
        threading.Thread(target=self._run, daemon=True).start()
        atexit.register(self._send)

    def emit(self, record):
        event = {
            "ts": int(record.created * 1000),
            "level": record.levelname,  # WARNING, CRITICAL… are mapped automatically
            "channel": self.channel or record.name,
            "text": record.getMessage(),
        }
        if record.exc_info:
            event["data"] = {"exception": self.formatException(record.exc_info)}
        try:
            self.q.put_nowait(event)
        except queue.Full:
            pass

    def _send(self):
        events = []
        while not self.q.empty() and len(events) < 500:
            events.append(self.q.get_nowait())
        if events:
            try:
                self.session.post(URL, json={"events": events}, timeout=5)
            except requests.RequestException:
                pass

    def _run(self):
        while True:
            time.sleep(self.flush_every)
            self._send()
```

```python
import logging
from logstrace import LogsTraceHandler

log = logging.getLogger("orders")
log.addHandler(LogsTraceHandler())
log.setLevel(logging.INFO)

log.info("Order %s paid %s %s by %s", order.id, order.total, order.currency, order.method)
```

::note
`record.getMessage()` fills in the `%s` values, so each message keeps the same fixed words and groups into one table.
::
